import { Formik, Form, Field, ErrorMessage } from "formik";
import React, { useEffect, useState } from "react";
import * as Yup from "yup";
import axios from 'axios';

const AnimeLoginPage = () => {
  const [isLogin, setIsLogin] = useState([]);

  useEffect(() => {
    const getUserData = JSON.parse(localStorage.getItem("AnimeUserData"));
    setIsLogin(getUserData || []);
  }, []);

  const validation = Yup.object({
    userName: Yup.string().min(5, "At least 5 characters").required("User name cannot be empty"),
    userPassword: Yup.string().min(5, "At least 5 characters").required("Password cannot be empty"),
    image: Yup.mixed()
      .required("An image is required")
    //   .test(
    //     "fileSize",
    //     "File size is too large. Max size is 2MB.",
    //     (value) => !value || (value && value.size <= 5)
    //     // (value) => !value || (value && value.size <= 2 * 1024 * 1024)
    //   )
    //   .test(
    //     "fileType",
    //     "Unsupported file format. Only jpg, png, jpeg are allowed.",
    //     (value) => !value || (value && ['image/jpeg', 'image/png', 'image/jpg'].includes(value.type))
    //   ),
  });

  const handleSubmit = async (values, { resetForm }) => {
    const formData = new FormData();
    formData.append('userName', values.userName);
    formData.append('userPassword', values.userPassword);
    formData.append('image', values.image);

    try {
      // Make the POST request to upload data
      const response = await axios.post('http://localhost:5000/api/items/imagePost', formData, {
        headers: {
          'Content-Type': 'multipart/form-data'
        }
      });
      console.log("Upload response:", response.data);
      
      // Update local state or perform other actions
      setIsLogin((prev) => [...prev, response.data]);
      localStorage.setItem("AnimeUserData", JSON.stringify([...isLogin, response.data]));

      resetForm();
    } catch (error) {
      console.error("Error uploading data:", error);
    }
  };

  const handleFileChange = (event, setFieldValue) => {
    setFieldValue("image", event.currentTarget.files[0]);
  };

  return (
    <>
      <h2>Anime Login Page</h2>
      <Formik
        initialValues={{
          userName: "",
          userPassword: "",
          image: null,
        }}
        validationSchema={validation}
        onSubmit={handleSubmit}
      >
        {({ setFieldValue }) => (
          <Form>
            <div>
              <label htmlFor="userName">User Name</label>
              <Field name="userName" type="text" />
              <ErrorMessage name="userName" component="div" style={{ color: "red" }} />
            </div>
            <div>
              <label htmlFor="userPassword">User Password</label>
              <Field name="userPassword" type="password" />
              <ErrorMessage name="userPassword" component="div" style={{ color: "red" }} />
            </div>
            <div>
              <label htmlFor="image">Upload Image</label>
              <input
                type="file"
                name="image"
                accept="image/*"
                onChange={(event) => handleFileChange(event, setFieldValue)}
              />
              <ErrorMessage name="image" component="div" style={{ color: "red" }} />
            </div>
            <button type="submit">Verify</button>
          </Form>
        )}
      </Formik>
    </>
  );
};

export default AnimeLoginPage;
